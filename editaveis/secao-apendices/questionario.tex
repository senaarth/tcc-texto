        \begin{itemize}
            \item ( ) Gerente de Produto
            \item ( ) \textit{Designer} de Produto
            \item ( ) \textit{Designer} Pedagócio
        \end{itemize}

        \item Há quanto tempo você trabalha na empresa?
        \begin{itemize}
            \item ( ) Entre 1 e 3 anos
            \item ( ) Entre 3 e 5 anos
            \item ( ) Mais de 5 anos
        \end{itemize}
    
        \item Você já tinha experiência prévia com experimentação ou alguma forma de desenvolvimento orientado a dados antes da implementação deste processo?
        \begin{itemize}
            \item ( ) Sim
            \item ( ) Não
        \end{itemize}
    
        \item Como você avalia a clareza com a qual processo de experimentação foi apresentado e implementado?  
        \textit{(Escala Likert de 1 a 5: 1 = Muito pouco claro, 5 = Muito claro)}
    
        \item O processo foi bem integrado ao fluxo de desenvolvimento da equipe e aos processos já existentes na empresa?  
        \textit{(Escala Likert de 1 a 5: 1 = Nada integrado, 5 = Muito bem integrado)}
    
        \item Você acredita que o uso de experimentação contribuiu para a qualidade das decisões sobre as versões do produto sendo avaliadas?  
        \textit{(Escala Likert de 1 a 5: 1 = Nada, 5 = Muito)}
    
        \item Quais benefícios você observou com a experimentação contínua? (Marque todas as opções que se aplicam)
        \begin{itemize}
            \item ( ) Maior confiabilidade nas tomadas de decisão sobre as versões do produto
            \item ( ) Redução de erros e retrabalho
            \item ( ) Melhor alinhamento com as necessidades do usuário
            \item ( ) Maior assertividade na identificação de problemas
            \item ( ) Outro: \_\_\_\_\_\_\_\_\_
        \end{itemize}
    
        \item Quais desafios você enfrentou ou percebeu durante a adoção da experimentação?
        \begin{itemize}
            \item ( ) Dificuldade de adaptação ao novo processo
            \item ( ) Falta de capacitação sobre a metodologia
            \item ( ) Tempo adicional necessário para coleta e análise de dados
            \item ( ) Resistência da equipe às mudanças
            \item ( ) Outro: \_\_\_\_\_\_\_\_\_
        \end{itemize}
    
        \item O tempo de tomada de decisão sobre uma nova versão foi afetado com o processo de experimentação?  
        \textit{(Escala Likert de 1 a 5: 1 = Foi aumentado, 5 = Foi reduzido)}

        \begin{itemize}
            \item ( ) Sim
            \item ( ) Não
            \item ( ) Outro: \_\_\_\_\_\_\_\_\_
        \end{itemize}

        \textit{(Campo aberto para resposta livre)}
